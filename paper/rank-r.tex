\begin{abstract}
Let $\Phi:M_d(\C)\to M_d(\C)$ be a completely positive map with a
transpose-symmetric Kraus representation.  Its Choi operator determines the
rank-two constant
\[
  \beta_2(\Phi)
  =\frac12\sup_{\substack{\norm z=1\\ \SR(z)\le2}}
       \ip z{J(\Phi)z}.
\]
We prove that, for every $r\ge1$,
\[
  \Phi\circ\tau+(r-1)\beta_2(\Phi)
       \left(\Delta-\frac1r\id\right)
\]
is $r$-positive, where $\tau$ is transposition and
$\Delta(X)=\Tr(X)I$.  The proof turns the negative spectrum of a partially
transposed rank-one operator into the edges of the complete graph $K_r$.
Regrouping by vertices costs its degree $r-1$, rather than its
$\binom r2$ edges.  Transpose symmetry protects one direction and is exactly
what produces the term $-\id/r$.

For $\Phi=\Lambda_U\otimes\Lambda_V$, with
$\Lambda_E(X)=\Tr(X)I-X^T$, a sharp double-skew singular-value estimate gives
$\beta_2\le1$, with equality when both local dimensions are at least two.  A
map identity then yields, for every operator $C$ of rank at most $r$ on
$U\otimes V$,
\[
  \norm{\Tr_U C}_2^2+\norm{\Tr_V C}_2^2
  \le r\norm C_2^2+\frac1r\abs{\Tr C}^2,
\]
with dimension-uniform coefficients that are sharp whenever one local
dimension is at least $r$ and the other is at least two.  A second family,
built from the edges of an arbitrary graph, has the same local constant and
attains the universal coefficient $r-1$ whenever the graph contains an
$r$-clique.  The Choi lifting chain through the partial-trace and graph
results, together with its sharpness witnesses, is formalized in Lean~4.
\end{abstract}

\section{The result behind the motivating inequality}

The motivating statement is a rank-sensitive relation among the two
marginals of an arbitrary bipartite matrix.  If
$C\in\cL(U\otimes V)$ and $\rank C\le r$, then
\begin{equation}
\label{eq:intro-marginal}
  \norm{\Tr_U C}_2^2+\norm{\Tr_V C}_2^2
  \le r\norm C_2^2+\frac1r\abs{\Tr C}^2.
\end{equation}
Here $\norm\cdot_2$ is the Hilbert--Schmidt norm and $\Tr_U$ means that
the $U$ factor is traced out.  Neither normality nor self-adjointness is
assumed.  The coefficients are sharp when one local dimension is at least
$r$ and the other is at least two.

Costa Rico proved \cref{eq:intro-marginal} for normal matrices and connected
its rank-two case with the two-copy distillability problem for Werner states
\cite{CostaRico}; Costa Rico and Wolf identified its extension beyond normal
matrices as an open problem \cite{CostaRicoWolf}.  Fu, Gao, and Park proved the
sharp rank-two double-skew estimate that supplies the local input used here
\cite{FuGaoPark2026}.  The arbitrary-$r$ inequality also appears, by a
balanced-factorization route, in Fraser, Huber, Pozsgay, and Vona
\cite{FraserHuberPozsgayVona2026}; the two lifting mechanisms are compared in
\cref{sec:balanced-route}.  Song and Chen
\cite{SongChen2026} and Bharti, Gajjala, and Haug
\cite{BhartiGajjalaHaug2026} prove the rank-two case and further consequences.
The contributions of this account are the general rank-two-to-$r$ lifting
theorem, the protected direction supplied by transpose symmetry, the graph
family showing universal sharpness, and the formal verification.  The
partial-trace inequality is the flagship application and a point of overlap
with the cited work.

Fix orthonormal bases whenever a transpose is used.  For a completely positive
map
\[
  \Phi(X)=\sum_a A_aXA_a^*
\]
on $\cL(H)$, let
\[
  J(\Phi)=\sum_a
  \ketbra{\vecop A_a}{\vecop A_a}
\]
be its Choi operator.  For a positive operator $X$ on a tensor product, set
\[
  \norm X_{S(k)}
  :=\sup_{\substack{\norm z=1\\\SR(z)\le k}}\ip z{Xz}.
\]
The convention $\vecop(\ketbra{x}{y})=x\otimes\overline y$ gives
$\SR(\vecop C)=\rank C$.

The \emph{Kraus space} of $\Phi$ is
\[
  \mathcal K_\Phi:=\operatorname{span}\{A_a\};
\]
equivalently, its vectorization is the range of $J(\Phi)$, so it is
independent of the Kraus representation.  We say that it has
\emph{even transpose parity} when $K^T=K$ for every
$K\in\mathcal K_\Phi$.

\paragraph{The rank-two Choi constant.}
For a completely positive map $\Phi$, define
\begin{equation}
\label{eq:beta}
  \beta_2(\Phi):=\frac12\norm{J(\Phi)}_{S(2)}.
\end{equation}

This depends on the map, not on a chosen Kraus representation.  If
$K_c=\sum_a c_aA_a$, then the projection duality for $S(k)$-norms
\cite{JohnstonKribs2010} gives the useful equivalent formula
\begin{equation}
\label{eq:beta-action}
  2\beta_2(\Phi)
  =\sup_{\norm c_2=1}
     \left(s_1(K_c)^2+s_2(K_c)^2\right),
\end{equation}
where $\norm c_2$ is the Euclidean coefficient norm and the singular values
are in decreasing order, padded by zeros beyond the matrix dimension.

\begin{theorem}[Rank-two lifting]
\label{thm:lift}
Let $\Phi:\cL(H)\to\cL(H)$ be completely positive.  For every integer
$r\ge1$, the map
\begin{equation}
\label{eq:unprotected}
  \Phi\circ\tau+(r-1)\beta_2(\Phi)\Delta
\end{equation}
is $r$-positive.

If $\mathcal K_\Phi$ has even transpose parity, then the sharper map
\begin{equation}
\label{eq:psi}
  \Psi_{r,\Phi}
  :=
  \Phi\circ\tau+(r-1)\beta_2(\Phi)
       \left(\Delta-\frac1r\id\right),
  \qquad \Delta(X)=\Tr(X)I,
\end{equation}
is $r$-positive.
\end{theorem}

Here a map is $r$-positive when its ampliation by the identity on
$M_r(\C)$ is positive.  Equivalently, its Choi operator has nonnegative
expectation on every vector of Schmidt rank at most $r$
\cite{Choi1975,JohnstonKribs2010}.

\begin{corollary}[Using an upper bound for the local constant]
\label{cor:beta-upper}
Under the protected hypothesis of \cref{thm:lift}, let $r\ge1$ be an integer
and let $\gamma\ge\beta_2(\Phi)$.  Then
\[
  \Phi\circ\tau+(r-1)\gamma
  \left(\Delta-\frac1r\id\right)
\]
is $r$-positive.
\end{corollary}

\begin{proof}
The map $\Delta-\frac1r\id$ is $r$-positive: for every
$C$ with $\rank C\le r$, its Choi quadratic form is
\[
  \norm C_2^2-\frac1r\abs{\Tr C}^2\ge0
\]
by the trace--rank inequality.  Add
$(r-1)(\gamma-\beta_2(\Phi))
(\Delta-\frac1r\id)$ to the protected map of \cref{thm:lift}.
\end{proof}

The theorem is not an induction on $r$, nor does it tensorize a rank-two
statement.  A partial transpose produces pair-indexed negative modes at level
$r$; the rank-two Choi bound controls each mode, and a complete-graph estimate
assembles them.  This distinction is useful when looking for new instances:
one needs a local two-vector estimate, not a recursive relation between
successive levels.

For a prospective application, four questions suffice.

\begin{enumerate}
\item Is $\Phi$ completely positive, and does its Kraus span have even
transpose parity?  Complete positivity gives \cref{eq:unprotected}; even
parity gives the protected pencil \cref{eq:psi}.
\item Can one bound the two-vector quantity
$s_1(K_c)^2+s_2(K_c)^2$?  This determines the universal sufficient
coefficient $(r-1)\beta_2(\Phi)$.
\item Is that coefficient necessary for the particular map?  This requires a
rank-$r$ Choi witness; saturation of the local two-vector bound alone need not
propagate through the complete-graph estimate.
\item Is there a map or contraction identity that turns the corrected Choi
form into the desired matrix expression?  Without one, the conclusion is
$r$-positivity, not the proposed matrix inequality.
\end{enumerate}

The improvement from \cref{eq:unprotected} to \cref{eq:psi} is a rank-one
protection, not a change in the complete-graph estimate.  The example in
\cref{sec:limits} shows that this improvement genuinely fails for an odd
Kraus space.

\section{The complete-graph proof}
\label{sec:proof}

The proof is included in the main text because its two structural steps---the
edge decomposition and the protected direction---are the information needed
to adapt the theorem.

Let $e_1,\dots,e_s$ be an orthonormal family in $H$, let
$q_1,\dots,q_s$ be the standard basis of $Q=\C^s$, and set
\[
  \delta_e=\sum_{i=1}^s e_i\otimes q_i,\qquad
  \rho_e=\ketbra{\delta_e}{\delta_e},\qquad
  \Pi_e=\frac1s\rho_e.
\]
The partial transpose of $\rho_e$ on $H$ is the flip between
$\operatorname{span}\{\overline e_i\}$ and $Q$.  Its negative spectral
projection is
\begin{equation}
\label{eq:pminus}
  P_-=\sum_{1\le i<j\le s}\ketbra{\eta_{ij}}{\eta_{ij}},
  \qquad
  \eta_{ij}=
  \frac{\overline e_i\otimes q_j-\overline e_j\otimes q_i}{\sqrt2}.
\end{equation}
Writing $P_+$ for the positive projection,
\begin{equation}
\label{eq:flip}
  \rho_e^{T_H}=P_+-P_-.
\end{equation}
The vectors $\eta_{ij}$ are naturally indexed by the edges of $K_s$.

\begin{proposition}[Negative-sector estimate]
\label{prop:negative}
For every completely positive $\Phi$ and every orthonormal family as above,
\begin{equation}
\label{eq:negative-unprotected}
  (\Phi\otimes\id_Q)(P_-)
  \preceqop (s-1)\beta_2(\Phi)I.
\end{equation}
If $\Phi$ has a transpose-symmetric Kraus representation, then
\begin{equation}
\label{eq:negative-protected}
  (\Phi\otimes\id_Q)(P_-)
  \preceqop (s-1)\beta_2(\Phi)(I-\Pi_e).
\end{equation}
\end{proposition}

\begin{proof}
Choose Kraus operators $\{A_a\}$ for $\Phi$.  For coefficients
$c=(c_a^{(ij)})$, indexed by edges $i<j$ and Kraus labels $a$, define
\[
  \mathcal Tc
  =\sum_{i<j}\sum_a c_a^{(ij)}
     (A_a\otimes I_Q)\eta_{ij},
  \qquad
  K^{(ij)}=\sum_a c_a^{(ij)}A_a.
\]
The Kraus formula and \cref{eq:pminus} give
\begin{equation}
\label{eq:ttstar}
  \mathcal T\mathcal T^*=(\Phi\otimes\id_Q)(P_-).
\end{equation}
Grouping the output by vertices rather than edges,
\begin{equation}
\label{eq:vertex-group}
  \mathcal Tc
  =\frac1{\sqrt2}\sum_{k=1}^s
   \left(
      \sum_{i<k}K^{(ik)}\overline e_i
      -\sum_{k<j}K^{(kj)}\overline e_j
   \right)\otimes q_k.
\end{equation}
There are $s-1$ incident terms at each vertex.  Cauchy--Schwarz at each
vertex therefore gives
\begin{align}
  \norm{\mathcal Tc}^2
  &\le\frac{s-1}{2}
    \sum_{i<j}
       \left(
         \norm{K^{(ij)}\overline e_i}^2+
         \norm{K^{(ij)}\overline e_j}^2
       \right).
\label{eq:vertex-cs}
\end{align}
For any orthonormal $x,y$ and $K=\sum_ac_aA_a$,
\begin{equation}
\label{eq:pair-action}
  \norm{Kx}^2+\norm{Ky}^2
  \le 2\beta_2(\Phi)\norm c_2^2.
\end{equation}
Indeed, if $P=\ketbra{x}{x}+\ketbra{y}{y}$, the Ky Fan variational
principle and the homogeneous form of \cref{eq:beta-action} give
\[
  \norm{Kx}^2+\norm{Ky}^2
  =\norm{KP}_2^2
  \le s_1(K)^2+s_2(K)^2
  \le2\beta_2(\Phi)\norm c_2^2.
\]
Substitution into \cref{eq:vertex-cs} yields
\begin{equation}
\label{eq:Tnorm}
  \norm{\mathcal Tc}^2
  \le(s-1)\beta_2(\Phi)\norm c_2^2.
\end{equation}
Together with \cref{eq:ttstar}, this proves
\cref{eq:negative-unprotected}.

Suppose now that $A_a^T=A_a$.  Then every $K^{(ij)}$ is symmetric, and
\begin{align*}
  \ip{\delta_e}{\mathcal Tc}
  &=
  \frac1{\sqrt2}\sum_{i<j}
   \left(
     \overline e_j^{\,T}K^{(ij)}\overline e_i
     -\overline e_i^{\,T}K^{(ij)}\overline e_j
   \right)=0.
\end{align*}
Thus $\ran\mathcal T\subseteq\delta_e^\perp$, and therefore
\[
  \mathcal T\mathcal T^*
  =(I-\Pi_e)\mathcal T\mathcal T^*(I-\Pi_e)
  \preceqop\norm{\mathcal T}^2(I-\Pi_e).
\]
Together with \cref{eq:Tnorm}, this proves
\cref{eq:negative-protected}.
\end{proof}

\begin{proof}[Proof of \cref{thm:lift}]
By \cref{eq:flip}, complete positivity of $\Phi$, and
\cref{eq:negative-unprotected},
\[
  (\Phi\otimes\id_Q)(\rho_e^{T_H})
  +(s-1)\beta_2(\Phi)I\succeqop0.
\]
Here
$I=(\Delta\otimes\id_Q)(\rho_e)$ because
$\Tr_H\rho_e=I_Q$.  Filtering on $Q$ produces every generating vector of
Schmidt rank at most $s$, and hence proves positivity on every corresponding
rank-one input.  Thus the map in
\cref{eq:unprotected} is $s$-positive at parameter $s$.  For a test vector
of Schmidt rank $s\le r$, increasing the parameter adds
$(r-s)\beta_2(\Phi)\Delta$, a completely positive map.  This proves the
unprotected assertion for every $r$.

Suppose now that the Kraus operators are transpose-symmetric.  Replacing
\cref{eq:negative-unprotected} by \cref{eq:negative-protected} gives
\begin{equation}
\label{eq:rho-positive}
  (\Phi\otimes\id_Q)(\rho_e^{T_H})
  +(s-1)\beta_2(\Phi)(I-\Pi_e)\succeqop0.
\end{equation}
Since $\Tr_H\rho_e=I_Q$ and $\rho_e=s\Pi_e$,
\[
  I-\Pi_e
  =
  \left[\left(\Delta-\frac1s\id\right)\otimes\id_Q\right](\rho_e).
\]
Hence \cref{eq:rho-positive} says
$(\Psi_{s,\Phi}\otimes\id_Q)(\rho_e)\succeqop0$.
Filtering on $Q$ proves the same assertion for every rank-one positive
operator whose generating vector has exact Schmidt rank $s$.

It remains to test the Choi operator of $\Psi_{r,\Phi}$ on a vector of Schmidt rank
$s\le r$.  Let $C$ be its inverse vectorization, so $\rank C=s$.  The
difference between the $\Psi_{r,\Phi}$ and $\Psi_{s,\Phi}$ Choi quadratic
forms is
\begin{equation}
\label{eq:rank-extension}
  (r-s)\beta_2(\Phi)
  \left(\norm C_2^2-\frac{\abs{\Tr C}^2}{rs}\right)\ge0,
\end{equation}
because $\abs{\Tr C}^2\le s\norm C_2^2$.  Thus the Choi operator of
$\Psi_{r,\Phi}$ is nonnegative on every vector of Schmidt rank at most
$r$, which is equivalent to $r$-positivity.
\end{proof}

The use of exact Schmidt rank $s$ in the final paragraph matters when
$r>\dim H$: an orthonormal $r$-frame in $H$ need not exist.  No padding
argument is required.

\section{The double-skew input}
\label{sec:double-skew}

For a finite-dimensional coordinate Hilbert space $E$, define
\[
  \Lambda_E(X)=\Tr(X)I_E-X^T.
\]
If $L_{ab}=E_{ab}-E_{ba}$, then
\begin{equation}
\label{eq:lambda-kraus}
  \Lambda_E(X)=\sum_{a<b}L_{ab}XL_{ab}^*.
\end{equation}
Thus $\Lambda_E$ is completely positive.  The Kraus operators of
$\Lambda_U\otimes\Lambda_V$ are $L_{ab}\otimes L_{cd}$, which are
transpose-symmetric because both factors are skew-symmetric.

The value of its Choi constant is the only map-specific input needed by
\cref{thm:lift}.

\begin{theorem}[Double-skew action bound]
\label{thm:double-skew}
Let
\[
  \mathfrak{so}(E)=\{L\in\cL(E):L^T=-L\}.
\]
For every
$K\in\mathfrak{so}(U)\otimes\mathfrak{so}(V)$ and every orthonormal
$x,y\in U\otimes V$,
\begin{equation}
\label{eq:double-skew}
  \norm{Kx}^2+\norm{Ky}^2
  \le\frac12\norm K_2^2.
\end{equation}
\end{theorem}

This is equivalent to the sharp Schmidt-rank-two projection estimate of
Fu, Gao, and Park \cite{FuGaoPark2026}.  The proof in
\cref{app:double-skew} adapts and reorganizes the argument of
\cite[\S2]{FuGaoPark2026}, using Autonne--Takagi directly; the following
roadmap isolates its delicate point.
The abstract lifting theorem consumes only the resulting local constant and
is independent of this concrete proof.
Tensor products of two skew matrices are symmetric under full transposition, so
Autonne--Takagi writes
\[
  K=\sum_i s_iw_iw_i^T.
\]
Let $\Pi$ be the Hilbert--Schmidt projection onto the double-skew space.
For every two-term Takagi truncation
$N=s_aw_aw_a^T+s_bw_bw_b^T$, a partial-transpose calculation gives
\[
  \ip N{N^{T_U}}_{\mathrm{HS}}\ge0,
  \qquad
  \ip N{\Pi N}_{\mathrm{HS}}\le\frac12\norm N_2^2.
\]
Projecting the truncation back against $K$ then forces
\[
  s_a^2+s_b^2\le\frac12\norm K_2^2
  \quad\text{for every pair }a\ne b.
\]
A two-coordinate weight argument turns these pair inequalities into
\cref{eq:double-skew}.  The nonnegativity after partial transpose is where
the tensor product of the two skew spaces enters.

\begin{corollary}[Exact local constant]
\label{cor:beta-one}
For $\Phi=\Lambda_U\otimes\Lambda_V$,
\[
  \beta_2(\Phi)\le1.
\]
Equality holds if $\dim U,\dim V\ge2$.  If either dimension is less than two,
then $\Phi=0$ and $\beta_2(\Phi)=0$.
\end{corollary}

\begin{proof}
The Kraus products in \cref{eq:lambda-kraus} are pairwise
Hilbert--Schmidt orthogonal and have squared norm $4$.  Thus, for
$K_c=\sum c_{ab,cd}L_{ab}\otimes L_{cd}$,
\[
  \norm{K_c}_2^2=4\norm c_2^2.
\]
\Cref{thm:double-skew} and \cref{eq:beta-action} give
$2\beta_2\le2$.  If both factors contain two distinct basis vectors, a
single Kraus product has four singular values equal to one; its two largest
squared singular values sum to $2$, proving equality.  In a space of
dimension below two there are no skew Kraus operators.
\end{proof}

\section{The partial-trace inequality}
\label{sec:marginal}

The general theorem becomes a marginal inequality only because
$\Lambda_U\otimes\Lambda_V$ satisfies a second, application-specific
identity.  Put
\[
  \mathcal R^E_{-a}=\Delta_E-a\id_E.
\]
Since $\Lambda_E\circ\tau_E=\Delta_E-\id_E$, a direct expansion in the
two-dimensional algebra $\operatorname{span}\{\Delta,\id\}$ gives, for every
integer $r\ge1$,
\begin{equation}
\label{eq:map-identity}
  (\Lambda_U\otimes\Lambda_V)\circ\tau
  +(r-1)\left(\Delta_{U\otimes V}-\frac1r\id\right)
  =
  r\,\mathcal R^U_{-1/r}\otimes\mathcal R^V_{-1/r}.
\end{equation}
By \cref{cor:beta-one}, $\beta_2(\Lambda_U\otimes\Lambda_V)\le1$ in every
dimension.  Applying \cref{cor:beta-upper} with $\gamma=1$ shows that the
left side of \cref{eq:map-identity}, and hence the map on the right, is
$r$-positive.  This includes the degenerate cases in which the sharp local
constant is zero.

\begin{theorem}[Sharp rank-$r$ partial-trace inequality]
\label{thm:marginal}
Let $r\ge1$ be an integer, and let $U,V$ be finite-dimensional complex
Hilbert spaces.  For every
$C\in\cL(U\otimes V)$ with $\rank C\le r$,
\begin{equation}
\label{eq:marginal}
  \norm{\Tr_U C}_2^2+\norm{\Tr_V C}_2^2
  \le r\norm C_2^2+\frac1r\abs{\Tr C}^2.
\end{equation}
If $C\ne0$ and $s=\rank C$, the stronger exact-rank form
\begin{equation}
\label{eq:exact-rank}
  \norm{\Tr_U C}_2^2+\norm{\Tr_V C}_2^2
  \le s\norm C_2^2+\frac1s\abs{\Tr C}^2
\end{equation}
holds.  Consequently, equality in \cref{eq:marginal} for nonzero $C$
requires $\rank C=r$.
\end{theorem}

\begin{proof}
The Choi quadratic form of the right side of \cref{eq:map-identity}, tested
on $\vecop C$, is
\begin{equation}
\label{eq:choi-contraction}
  r\norm C_2^2+\frac1r\abs{\Tr C}^2
  -\norm{\Tr_U C}_2^2-\norm{\Tr_V C}_2^2.
\end{equation}
This follows by expanding the tensor product: the four terms
$\Delta\otimes\Delta$, $\Delta\otimes\id$, $\id\otimes\Delta$, and
$\id\otimes\id$ contract respectively to
$\norm C_2^2$, the two marginal Hilbert--Schmidt norms, and
$\abs{\Tr C}^2$, with the coefficients in
\cref{eq:map-identity}.  Since $r$-positivity is equivalent to
nonnegativity of the Choi quadratic form on inverse vectorizations of rank at
most $r$, \cref{eq:choi-contraction} is nonnegative.

Apply the result first with $r=s=\rank C$ to obtain
\cref{eq:exact-rank}.  For $s\le r$,
\begin{align*}
 &r\norm C_2^2+\frac1r\abs{\Tr C}^2
 -\left(s\norm C_2^2+\frac1s\abs{\Tr C}^2\right)\\
 &\hspace{28mm}=(r-s)
   \left(\norm C_2^2-\frac{\abs{\Tr C}^2}{rs}\right)\ge0.
\end{align*}
The trace--rank inequality $\abs{\Tr C}^2\le s\norm C_2^2$ proves the
last step.  If $C\ne0$ and $s<r$, both factors in the last display are
strictly positive, so equality is impossible.
\end{proof}

\begin{proposition}[Optimal coefficients]
\label{prop:optimal}
Let $r\ge1$.  Suppose $\dim U\ge r$ and $\dim V\ge2$, or conversely.  In an inequality
\[
  \norm{\Tr_U C}_2^2+\norm{\Tr_V C}_2^2
  \le a\norm C_2^2+b\abs{\Tr C}^2
  \qquad(\rank C\le r),
\]
one must have $a\ge r$.  Once $a=r$ is fixed, one must have
$b\ge1/r$.
\end{proposition}

\begin{proof}
Let $P_r$ be a rank-$r$ projection on $U$ and let $v,w\in V$ be unit
vectors.  For
\[
  C=P_r\otimes\ketbra{v}{w}
\]
one has
\[
\begin{array}{c@{\qquad}c}
  \norm{\Tr_U C}_2^2=r^2,
  &\norm{\Tr_V C}_2^2=r\abs{\ip wv}^2,\\
  \norm C_2^2=r,
  &\abs{\Tr C}^2=r^2\abs{\ip wv}^2.
\end{array}
\]
Taking $v\perp w$ forces $a\ge r$.  With $a=r$, taking $v=w$ forces
$b\ge1/r$.  The same matrices attain equality in \cref{eq:marginal}.
\end{proof}

\subsection{A complementary balanced-factorization lift}
\label{sec:balanced-route}

Fraser, Huber, Pozsgay, and Vona prove \cref{eq:marginal} directly from a
balanced rank decomposition \cite[Lem.~7.1 and \S8]{FraserHuberPozsgayVona2026}.
Their proof exposes a second lifting principle.  The following formulation is
the abstraction of the two properties used in their argument.

\begin{proposition}[Balanced-polarization lift]
\label{prop:balanced-lift}
Let $\mathcal Y$ be a finite-dimensional complex Hilbert space and let
$\mathcal L:\cL(H)\to\mathcal Y$ be linear.  Suppose that, for all
$x,y,u,v,s,t\in H$,
\begin{align}
\label{eq:balanced-rank-one}
  \norm{\mathcal L(xy^*)}^2
  &\le \norm x^2\norm y^2+\abs{y^*x}^2,\\
\label{eq:crossed-polarization}
  \ip{\mathcal L(uv^*)}{\mathcal L(st^*)}
  &=\ip{\mathcal L(tv^*)}{\mathcal L(su^*)}.
\end{align}
If $C\ne0$ and $q=\rank C$, then
\begin{equation}
\label{eq:balanced-lift}
  \norm{\mathcal L(C)}^2
  \le q\norm C_2^2+\frac1q\abs{\Tr C}^2.
\end{equation}
Consequently the right side may be replaced by
$r\norm C_2^2+\abs{\Tr C}^2/r$ whenever $q\le r$.
\end{proposition}

\begin{proof}
Choose a balanced factorization
\[
  C=\sum_{i=1}^q x_i y_i^*,\qquad
  X^*X=Y^*Y=G=(g_{ij}),\qquad
  y_i^*x_i=\tau:=\frac{\Tr C}{q}.
\]
It is obtained by splitting the singular values equally between $X$ and $Y$,
then applying the same unitary to their columns so that the diagonal of
$Y^*X$ is constant.  The latter step is the Parker--Fillmore
constant-diagonal theorem \cite{Fillmore1969,Parker1948}.  Put
$\delta_i=g_{ii}=\norm{x_i}^2=\norm{y_i}^2$.

The diagonal terms in the expansion of
$\norm{\sum_i\mathcal L(x_i y_i^*)}^2$ are at most
$\delta_i^2+\abs\tau^2$ by \cref{eq:balanced-rank-one}.  For $i\ne j$,
\cref{eq:crossed-polarization}, Cauchy--Schwarz, and another application of
\cref{eq:balanced-rank-one} give
\[
  \abs{\ip{\mathcal L(x_i y_i^*)}{\mathcal L(x_j y_j^*)}}
  \le \delta_i\delta_j+\abs{g_{ij}}^2.
\]
Therefore
\[
  \norm{\mathcal L(C)}^2
  \le\left(\sum_i\delta_i\right)^2
      +q\abs\tau^2+2\sum_{i<j}\abs{g_{ij}}^2.
\]
On the other hand,
\[
  \norm C_2^2=\Tr(G^2)
  =\sum_i\delta_i^2+2\sum_{i<j}\abs{g_{ij}}^2.
\]
Subtracting the preceding upper bound from the right side of
\cref{eq:balanced-lift} gives the explicit defect bound
\begin{equation}
\label{eq:balanced-defect}
  q\norm C_2^2+\frac1q\abs{\Tr C}^2-\norm{\mathcal L(C)}^2
  \ge \sum_{i<j}(\delta_i-\delta_j)^2
      +2(q-1)\sum_{i<j}\abs{g_{ij}}^2
  \ge0.
\end{equation}
The extension from $q$ to $r$ is the same trace--rank calculation used after
\cref{eq:choi-contraction}.
\end{proof}

For the partial-trace application, take
\[
  \mathcal L(C)=(\Tr_U C,\Tr_V C)
\]
in the Hilbert direct sum.  Then \cref{eq:balanced-rank-one} is the rank-one
partial-trace inequality; its defect is the expectation of
$(I-F_U)\otimes(I-F_V)=4P_U^-\otimes P_V^-$.  \Cref{eq:crossed-polarization}
is the four-state marginal identity.  Thus this
route and the Choi lift use the same double-antisymmetric geometry but extract
different information from it.

The mechanisms are not reformulations of one another.  The balanced lift
requires the rigid four-linear symmetry
\cref{eq:crossed-polarization}, but neither complete positivity nor a Choi
norm; it gives a direct norm inequality.  \Cref{thm:lift} requires a
rank-two Choi bound and, for the protected term, even transpose parity, but no
crossed identity; it returns an $r$-positive map.  Both arguments nevertheless
contain the same complete graph.  In \cref{eq:balanced-defect} the first sum is
the Dirichlet energy of $K_q$, while in \cref{eq:vertex-cs} its degree
$q-1$ appears by grouping edge modes at vertices.  The constant diagonal
isolates the trace direction in the balanced proof; range orthogonality to
$\delta_e$ protects the corresponding direction in the Choi proof.

There is also a common operator language.  Let
$\Gamma=\mathcal L^*\mathcal L$ on the Hilbert--Schmidt space and
$\Omega=\sum_i e_i\otimes\overline e_i$.  In matrix-unit coordinates,
\cref{eq:crossed-polarization} is exactly the reshuffling symmetry
\[
  \Gamma_{ij,kl}=\Gamma_{lj,ki}.
\]
The rank-one hypothesis is block positivity of
$I+\ketbra{\Omega}{\Omega}-\Gamma$ on product vectors; the conclusion is
block positivity of
$rI+\ketbra{\Omega}{\Omega}/r-\Gamma$ on vectors of Schmidt rank at most $r$.
Thus the balanced argument is itself a low-rank positivity lift, based on a
crossing-fixed positive Gram operator and a level-one input rather than a CP
Choi operator and a level-two input.

\begin{corollary}[Spectral stability]
\label{cor:spectral-stability}
Let $C\in\cL(U\otimes V)$ have rank $q\ge2$, let
$\sigma_1,\dots,\sigma_q$ be its nonzero singular values, and put
$\overline\sigma=q^{-1}\sum_i\sigma_i$.  If
\[
  D_q(C):=
  q\norm C_2^2+\frac1q\abs{\Tr C}^2
  -\norm{\Tr_U C}_2^2-\norm{\Tr_V C}_2^2,
\]
then
\begin{equation}
\label{eq:spectral-stability}
  \sum_{i=1}^q(\sigma_i-\overline\sigma)^2
  \le\frac{D_q(C)}{q-1}.
\end{equation}
\end{corollary}

\begin{proof}
In the balanced construction, $G$ is unitarily similar to
$\operatorname{diag}(\sigma_1,\dots,\sigma_q)$.  If
$\overline\delta=q^{-1}\sum_i\delta_i=\overline\sigma$, then
\[
  \norm{G-\overline\delta I}_2^2
  =\frac1q\sum_{i<j}(\delta_i-\delta_j)^2
    +2\sum_{i<j}\abs{g_{ij}}^2.
\]
Multiplying by $q-1$ and comparing with
\cref{eq:balanced-defect} proves \cref{eq:spectral-stability}.
\end{proof}

\section{A second instance: graph Kraus dictionaries}
\label{sec:graphs}

The next family does two jobs.  It shows that the lifting theorem applies
away from the partial-trace identity, and it proves that
$(r-1)\beta_2$ is an attained universal coefficient rather than merely a
proof artifact.

Let $G=(V,E)$ be a nonempty simple graph on $V=\{1,\dots,m\}$.  For
$e=\{i,j\}$, $i<j$, put
\[
  L_e=E_{ij}-E_{ji}\in M_m,\qquad
  J=E_{01}-E_{10}\in M_2,\qquad
  A_e=L_e\otimes J,
\]
and define
\begin{equation}
\label{eq:graph-map}
  \Phi_G(X)=\sum_{e\in E}A_eXA_e^*.
\end{equation}
Every $A_e$ is real symmetric.  For
$K_c=\sum_ec_eA_e=L_c\otimes J$, the nonzero singular values of the
skew matrix $L_c$ occur in equal pairs.  Since
$\norm{L_c}_2^2=2\norm c_2^2$ and $J$ is unitary,
\[
  s_1(K_c)^2+s_2(K_c)^2
  =2s_1(L_c)^2\le2\norm c_2^2.
\]
A coefficient vector supported on one edge attains equality.  Hence
\begin{equation}
\label{eq:graph-beta}
  \beta_2(\Phi_G)=1.
\end{equation}

For an integer $R\ge1$, set
\[
  \Theta^G_{R,\lambda}
  =\Phi_G\circ\tau+
    \lambda\left(\Delta-\frac1R\id\right),
\]
and set
\[
  \lambda_R^*(G)
  :=\inf\{\lambda\ge0:\Theta^G_{R,\lambda}\text{ is $R$-positive}\}.
\]

\begin{theorem}[Induced-density obstruction and clique sharpness]
\label{thm:graph}
For every nonempty $S\subseteq V$ with $\abs S\le R$,
\begin{equation}
\label{eq:graph-lower}
  \frac{2\abs{E(G[S])}}{\abs S}
  \le\lambda_R^*(G)\le R-1.
\end{equation}
In particular, if $G$ contains an $R$-clique, then
\[
  \lambda_R^*(G)=R-1.
\]
\end{theorem}

\begin{proof}
The upper bound is \cref{thm:lift} and
\cref{eq:graph-beta}.  For the lower bound, take
$C=P_S\otimes E_{01}$, where
$P_S=\sum_{i\in S}E_{ii}$.  Then
\[
  \rank C=\abs S,\qquad \Tr C=0,\qquad\norm C_2^2=\abs S.
\]
Because the $A_e$ are symmetric, the Choi convention gives
\begin{equation}
\label{eq:graph-choi-pairing}
  \ip{\vecop C}{J(\Phi_G\circ\tau)\vecop C}
  =
  \sum_{e\in E}
  \ip{A_eC}{(A_eC)^T}_{\mathrm{HS}}.
\end{equation}
If both endpoints of $e$ lie in $S$, then
\[
  A_eC=-L_e\otimes E_{11},
  \qquad
  (A_eC)^T=-A_eC,
\]
so its contribution to \cref{eq:graph-choi-pairing} is
$-\norm{L_e}_2^2=-2$.  If exactly one endpoint lies in $S$, then
$L_eP_S$ is, up to sign, a single off-diagonal matrix unit; its
Hilbert--Schmidt pairing with its transpose is zero.  If neither endpoint
lies in $S$, then $A_eC=0$.  The correction contributes
$\lambda\abs S$, and nonnegativity forces \cref{eq:graph-lower}.  If $S$
is an $R$-clique, its induced average degree is $R-1$, matching the
universal upper bound.
\end{proof}

There is a useful warning in the most natural compression of this graph
family.  Let
\[
  \mathcal S_G=\operatorname{span}\{E_{ii}\}
  +\operatorname{span}\{E_{ij}:\{i,j\}\in E\},
\]
and write $p=\ketbra00$, $q=\ketbra11$.  Compressing input
$X\otimes q$ to the $p$ output corner recovers the graph reduction map,
but it annihilates the input itself.  Under this compression,
\[
  \Delta_{2m}-\frac1R\id\quad\longmapsto\quad\Delta_m.
\]
Thus the one-layer compression preserves the unprotected theorem
\cref{eq:unprotected} and discards precisely the $-\id/R$ term.  Keeping both
diagonal qubit layers repairs the loss.  The lesson is broader than this
example: a coefficient-space reduction must preserve the protected vector,
not only the Kraus action.

\section{Sharp boundaries of the method}
\label{sec:limits}

\subsection{Odd transpose parity}

Transposition grades the ambient matrix space through
$\C^d\otimes\C^d=\operatorname{Sym}^2\C^d\oplus\wedge^2\C^d$.
A Kraus space need not itself be transpose-invariant.  The protected
hypothesis selects the even summand: symmetric matrices have even degree,
skew-symmetric matrices odd degree, and tensor-product degrees add modulo
two.  The cancellation in the proof of
\cref{eq:negative-protected} occurs exactly in even degree.

The even-parity hypothesis cannot be dropped from the universal protected
conclusion.
Let $H=\C^2$ and
\[
  A=E_{01}-E_{10},\qquad \Phi(X)=AXA^*.
\]
The one-dimensional Kraus space is skew.  At $s=2$, using the standard
bases in \cref{eq:pminus},
\[
  (A\otimes I)\eta_{12}=-\frac1{\sqrt2}\delta_e,
  \qquad
  (\Phi\otimes\id)(P_-)=\Pi_e.
\]
Consequently,
\[
  (\Phi\otimes\id)(P_-)
  \preceqop\lambda(I-\Pi_e)
\]
fails for every finite $\lambda$: the right side vanishes on
$\delta_e$ and the left side does not.  The unprotected bound is sharp,
since $\Pi_e\preceqop I$ and $\beta_2(\Phi)=1$.

\subsection{A Choi constant is not a contraction identity}

\Cref{thm:lift} returns an $r$-positive map.  It does not say which concrete
quadratic form that map represents.  The partial traces in
\cref{eq:marginal} enter through \cref{eq:map-identity} and
\cref{eq:choi-contraction}, not through $\beta_2$.  The zero map makes the
logical separation immediate: it satisfies the theorem with
$\beta_2=0$, but carries no information about either marginal.  Any proposed
new application therefore has two independent obligations:
compute the local Choi constant and identify the downstream quadratic form.

\subsection{The universal coefficient need not be the exact coefficient}

The vertex Cauchy--Schwarz step \cref{eq:vertex-cs} discards nonnegative
variance terms.  It follows that $(r-1)\beta_2(\Phi)$ is a universal
sufficient coefficient, not a formula for the optimum of every map.  The
clique case of \cref{thm:graph} shows that no smaller universal coefficient
is possible.  For a particular Kraus dictionary without an appropriate
witness, its exact correction threshold remains a separate extremal problem.

\subsection{Higher tensor powers}

An even tensor product of skew Kraus spaces still has even transpose parity,
so \cref{thm:lift} applies whenever its $\beta_2$ is known.  This gives
alternating marginal inequalities for higher even tensor powers.  Two
features of the two-factor application do not automatically persist:

\begin{enumerate}
\item The constants
$\beta_2(\Lambda_{U_1}\otimes\cdots\otimes\Lambda_{U_{2m}})$ are not
determined by \cref{thm:double-skew} when $m\ge2$.
\item The collapse in \cref{eq:map-identity} is special to two factors.  If
$S=\Delta-\id$ and one tries to match
$S^{\otimes n}+(r-1)\Delta^{\otimes n}-t\id$ with a scalar multiple of
$(S+t\id)^{\otimes n}$, the terms containing $n-1$ copies of $S$ force
$rt=r-1$, while those containing $n-2$ copies force
$r-1=(r-1)^2/r$.  For $n\ge3$, these equations are compatible only at
$r=1$.
\end{enumerate}

The lifting theorem therefore survives in higher even degree, while the
particularly simple tensor-product reduction map does not.

\section{Formal verification and scope}
\label{sec:formal}

The finite-coordinate development proves \cref{thm:double-skew}, both the
protected and unprotected parts of \cref{thm:lift},
\cref{cor:beta-upper}, \cref{thm:marginal,thm:graph}, the exact-rank
extension, both optimality witnesses, and the parity counterexample.  The
accompanying repository
\cite{RankRRepository} is cited at the immutable revision inspected for this
manuscript.  It also proves the Autonne--Takagi factorization used in
\cref{app:double-skew}.  The main declarations contain no \texttt{sorry};
their reported axiom surface is \texttt{propext},
\texttt{Classical.choice}, and \texttt{Quot.sound}.

The formal statements use matrices over finite coordinate types.
They prove the coordinate identities and rank bounds used in this paper,
including $\SR(\vecop C)=\rank C$.  The transport from abstract
finite-dimensional Hilbert spaces to chosen orthonormal coordinates is not a
separate formal theorem.  Accordingly, the kernel checks the deduction once
those coordinate definitions are accepted; the correspondence between those
definitions and the basis-free prose remains an interface to inspect.

For the balanced-polarization route in \cref{sec:balanced-route}, the cited
revision checks the generic lift, its complete-graph defect, the sharp
rank-one marginal estimate, the crossed-polarization identity, and the
unconditional partial-trace specialization.  It
also constructs the exact-rank equal-Gram factorization by spectral mixing and
fourth-root rescaling, and checks that constant-diagonal unitary mixing
balances it without changing the represented matrix.  The trace-entry
construction and Parker--Fillmore constant-diagonal induction are formalized
as part of that route.

\section{Proof of the double-skew action bound}
\label{app:double-skew}

We prove \cref{thm:double-skew}, adapting the argument of
\cite[\S2]{FuGaoPark2026} and reorganizing it around the Autonne--Takagi
factorization.  Relative to the fixed product basis, let
$T_U$ and $T_V$ denote the two partial transposes on
$\cL(U\otimes V)$.  They are commuting self-adjoint involutions for the
Hilbert--Schmidt inner product, and
\[
  \Pi=\frac12(1-T_U)\frac12(1-T_V)
\]
is the orthogonal projection onto
$\mathfrak{so}(U)\otimes\mathfrak{so}(V)$.  Hence $\Pi K=K$.
Moreover $K^T=K$, because each elementary tensor contains two skew factors.

Autonne--Takagi factorization \cite[Cor.~4.4.4(c)]{HornJohnson} gives
\begin{equation}
\label{eq:takagi}
  K=\sum_i s_iw_iw_i^T,
\end{equation}
where the $w_i$ form an orthonormal basis, $s_i\ge0$, and zero values of
$s_i$ are retained.  The matrices $w_iw_i^T$ are Hilbert--Schmidt
orthonormal.

We first establish the two-term estimate.  Write $M_u$ for the coefficient
matrix of $u\in U\otimes V$.  Direct contraction of the two product indices
gives
\begin{equation}
\label{eq:flip-trace}
  \ip{uu^T}{(vv^T)^{T_U}}_{\mathrm{HS}}
  =\Tr\bigl((M_u^*M_v)^2\bigr).
\end{equation}
For orthonormal $w_1,w_2$, let
\[
  N=s_1w_1w_1^T+s_2w_2w_2^T,\qquad
  c_i=\norm{M_i^*M_i}_2.
\]
Expanding \cref{eq:flip-trace},
\[
  \ip N{N^{T_U}}_{\mathrm{HS}}
  =s_1^2c_1^2+
    2s_1s_2\operatorname{Re}\Tr((M_1^*M_2)^2)+s_2^2c_2^2.
\]
The Hilbert--Schmidt Cauchy--Schwarz inequalities
\[
  \abs{\Tr(X^2)}\le\norm X_2^2,
  \qquad
  \norm{A^*B}_2^2
  =\ip{AA^*}{BB^*}_{\mathrm{HS}}
  \le\norm{A^*A}_2\norm{B^*B}_2
\]
bound the middle term from below by $-2s_1s_2c_1c_2$.  Therefore
\begin{equation}
\label{eq:partial-transpose-positive}
  \ip N{N^{T_U}}_{\mathrm{HS}}
  \ge(s_1c_1-s_2c_2)^2\ge0.
\end{equation}
Since $N^T=N$, one has $N^{T_U}=N^{T_V}$, and expansion of $\Pi$ gives
\begin{equation}
\label{eq:projection-half}
  \ip N{\Pi N}_{\mathrm{HS}}
  =\frac12\left(\norm N_2^2-
        \ip N{N^{T_U}}_{\mathrm{HS}}\right)
  \le\frac12\norm N_2^2.
\end{equation}

Now fix $a\ne b$ and truncate \cref{eq:takagi} to
\[
  K_{ab}=s_aw_aw_a^T+s_bw_bw_b^T,\qquad
  A=s_a^2+s_b^2.
\]
Orthogonality gives
$\norm{K_{ab}}_2^2=\ip K{K_{ab}}_{\mathrm{HS}}=A$.  Since $\Pi$ is a
self-adjoint projection and $\Pi K=K$,
\[
  A=\ip K{\Pi K_{ab}}_{\mathrm{HS}}
  \le\norm K_2\norm{\Pi K_{ab}}_2.
\]
By \cref{eq:projection-half},
$\norm{\Pi K_{ab}}_2^2
=\ip{K_{ab}}{\Pi K_{ab}}_{\mathrm{HS}}\le A/2$.
Thus $A^2\le A\norm K_2^2/2$, and cancellation when $A>0$ yields
\begin{equation}
\label{eq:all-pairs}
  s_a^2+s_b^2\le\frac12\norm K_2^2
  \qquad(a\ne b).
\end{equation}

Finally, \cref{eq:takagi} implies
$K^*K=\sum_i s_i^2\ketbra{\overline w_i}{\overline w_i}$.
For orthonormal $x,y$, set
\[
  \alpha_i=\abs{\ip{\overline w_i}{x}}^2+
           \abs{\ip{\overline w_i}{y}}^2.
\]
Then $0\le\alpha_i\le1$ and $\sum_i\alpha_i=2$.  The polytope
\[
  \{\alpha:0\le\alpha_i\le1,\ \sum_i\alpha_i=2\}
\]
is the convex hull of the vectors having two coordinates equal to one and
the rest zero.  Hence \cref{eq:all-pairs} gives
\[
  \norm{Kx}^2+\norm{Ky}^2
  =\sum_i\alpha_is_i^2
  \le\max_{a\ne b}(s_a^2+s_b^2)
  \le\frac12\norm K_2^2.
\]
This proves \cref{thm:double-skew}.

\section{Elementary Choi contractions}
\label{app:choi}

For completeness, we record the four contractions used in
\cref{eq:choi-contraction}.  With the vectorization convention of
\cref{sec:proof},
\begin{align*}
  \ip{\vecop C}{J(\Delta_U\otimes\Delta_V)\vecop C}
    &=\norm C_2^2,\\
  \ip{\vecop C}{J(\id_U\otimes\Delta_V)\vecop C}
    &=\norm{\Tr_U C}_2^2,\\
  \ip{\vecop C}{J(\Delta_U\otimes\id_V)\vecop C}
    &=\norm{\Tr_V C}_2^2,\\
  \ip{\vecop C}{J(\id_U\otimes\id_V)\vecop C}
    &=\abs{\Tr C}^2.
\end{align*}
They follow by expanding $C$ in matrix units and matching indices.  Their
order depends only on the convention that $\Tr_U$ traces out $U$; interchanging
$U,V$ interchanges the middle two lines and leaves every result unchanged.
